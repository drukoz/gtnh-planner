// TEMPORARY SP-M (RUK-3058) probe. NEVER COMMIT. Writes rich per-row data, the benchTable, and timings to paths in env.
import { it, expect } from 'vitest';
import { writeFileSync } from 'node:fs';
import { CandidateKind } from '../../../src/shared/types/layout.ts';
import { layoutLine } from '../../../src/server/layout/layout.ts';
import type { LogisticsMode } from '../../../src/shared/types/plan.ts';
import { benchLines, lengthOf, runBench, benchTable, type LengthReport } from '../../helpers/layout-bench.ts';
import { layoutPack, makeProject } from '../../helpers/layout-fixtures.ts';
import { lossPlan, lossScan, lossGround } from '../../helpers/power-loss.ts';

const MODES: readonly LogisticsMode[] = ['direct', 'routed', 'ae2', 'enderio'];

interface RichRow {
  name: string; mode: LogisticsMode;
  kind: number; westUnits: number; laid: number;
  score: { unwired: number; unreachable: number; handTransformers: number; unpowered: number; total: number; lossEUt: number; volume: number };
  length: LengthReport;
  codes: string[];
}

const median = (xs: number[]): number => {
  const s = [...xs].sort((a, b) => a - b);
  const m = s.length >> 1;
  return s.length % 2 ? s[m]! : (s[m - 1]! + s[m]!) / 2;
};

async function layOne(plan: ReturnType<typeof lossPlan>, mode: LogisticsMode, modeIdx: number): Promise<{ ms: number; laid: number }> {
  const lines = benchLines();
  const line = lines[modeIdx]!;
  const t0 = performance.now();
  const layout = await layoutLine(layoutPack(), line.plan, line.scan(mode), makeProject({ logistics: mode }), line.deps);
  const ms = performance.now() - t0;
  return { ms, laid: layout.search?.candidates.length ?? 0 };
}

it('probe', async () => {
  const jsonPath = process.env.SPM_JSON;
  const benchPath = process.env.SPM_BENCH;
  const timePath = process.env.SPM_TIME;
  expect(jsonPath ?? benchPath ?? timePath, 'a probe output path is required').toBeTruthy();

  if (benchPath !== undefined) {
    const rows = await runBench();
    writeFileSync(benchPath, benchTable(rows) + '\n');
  }

  if (jsonPath !== undefined) {
    const rich: RichRow[] = [];
    for (const line of benchLines()) {
      for (const mode of MODES) {
        const layout = await layoutLine(layoutPack(), line.plan, line.scan(mode), makeProject({ logistics: mode }), line.deps);
        const search = layout.search!;
        const cand = search.candidates[search.chosen]!;
        rich.push({
          name: line.name, mode, kind: cand.kind, westUnits: cand.westUnits, laid: search.candidates.length,
          score: { unwired: cand.unwired, unreachable: cand.unreachable, handTransformers: cand.handTransformers, unpowered: cand.unpowered, total: cand.total, lossEUt: cand.lossEUt, volume: cand.volume },
          length: { name: line.name, ...lengthOf(layout) },
          codes: [...new Set(layout.warnings.map((w) => w.code))].sort(),
        });
      }
    }
    writeFileSync(jsonPath, JSON.stringify(rich, null, 1));
  }

  if (timePath !== undefined) {
    const benchMs: number[] = [];
    for (let i = 0; i < 3; i++) {
      const t0 = performance.now();
      await runBench();
      benchMs.push(performance.now() - t0);
    }
    const perLine: Array<{ name: string; mode: LogisticsMode; laid: number; ms: number }> = [];
    // loss lines by size (direct + ae2)
    for (const n of [10, 20, 40, 60, 80]) {
      for (const mode of ['direct', 'ae2'] as const) {
        const pack = layoutPack();
        const plan = lossPlan(n);
        let best = { ms: Infinity, laid: 0 };
        for (let r = 0; r < 3; r++) {
          const t0 = performance.now();
          const layout = await layoutLine(pack, plan, lossScan(mode), makeProject({ logistics: mode }), lossGround());
          const ms = performance.now() - t0;
          const laid = layout.search?.candidates.length ?? 0;
          if (ms < best.ms) best = { ms, laid };
        }
        perLine.push({ name: `loss-${n}`, mode, laid: best.laid, ms: Math.round(best.ms * 100) / 100 });
      }
    }
    // the largest mixed line: the one with the most machines
    const mixed = benchLines().filter((l) => l.name.startsWith('mixed-'));
    const biggest = mixed.reduce((a, b) => (units(b.plan) >= units(a.plan) ? b : a));
    for (const mode of MODES) {
      let best = { ms: Infinity, laid: 0 };
      for (let r = 0; r < 3; r++) {
        const t0 = performance.now();
        const layout = await layoutLine(layoutPack(), biggest.plan, biggest.scan(mode), makeProject({ logistics: mode }), biggest.deps);
        const ms = performance.now() - t0;
        const laid = layout.search?.candidates.length ?? 0;
        if (ms < best.ms) best = { ms, laid };
      }
      perLine.push({ name: `${biggest.name}(${units(biggest.plan)})`, mode, laid: best.laid, ms: Math.round(best.ms * 100) / 100 });
    }
    writeFileSync(timePath, JSON.stringify({ runBenchMs: benchMs, runBenchMedian: Math.round(median(benchMs)), perLine }, null, 1));
  }
}, 900_000);

function units(plan: { steps: Array<{ count: number }> }): number {
  return plan.steps.reduce((n, s) => n + s.count, 0);
}
